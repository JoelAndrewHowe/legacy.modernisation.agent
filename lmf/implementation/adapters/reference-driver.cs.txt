using System.Text.Json;
using System.Text.Json.Nodes;
using Orvex.Business.Validation;
using Orvex.Messages.Faults;
using Orvex.TransferObjects;

var options = new JsonSerializerOptions { PropertyNameCaseInsensitive = true };
var scenarios = JsonNode.Parse(File.ReadAllText(args[0])).AsArray();
var fixtures = JsonNode.Parse(File.ReadAllText(args[1]));
ReferenceClock.Today = DateTime.Parse(fixtures["today"].GetValue<string>());
var birth = DateTime.Parse(fixtures["dateOfBirth"].GetValue<string>());
var observations = new JsonArray();
foreach (var scenario in scenarios)
{
    var input = scenario["input"];
    var result = input.Deserialize<KdpExaminationResultTO>(options);
    var errors = new List<string>();
    var warnings = new List<string>();
    var observed = new JsonObject();
    var validator = new KdpExaminationValidator();
    try
    {
        switch (scenario["when"].GetValue<string>())
        {
            case "save":
                validator.ValidateOnSave(result, birth, null, new());
                // Detached observation only, not a claim of database read-back.
                observed["saved"] = input.DeepClone();
                break;
            case "head-field": validator.ValidateHeadCircumference(result.HeadCircumference.Value); break;
            case "duplicate-date":
                validator.ValidateExamCodeForDate(new() { new() { ExaminationCode = "KD03", ExaminationDate = result.ExaminationDate, PkChildExaminationId = 42 } }, 0, "KD03", result.ExaminationDate);
                break;
            case "duplicate-code":
                validator.ValidateExamCodeForDuplicates(new() { new() { ExaminationCode = "KD03", PkChildExaminationId = 42 } }, "KD03", 0);
                break;
            case "consent": validator.ValidateKdpConsent(result.ClientId, false); break;
            case "age":
                var age = new ExtractedAge { dateOfBirth = birth, recommendedAgeRangeInWeeks = new(() => new(fixtures["scheduleWeeks"][0].GetValue<double>(), fixtures["scheduleWeeks"][1].GetValue<double>())) };
                age.CheckAgeWithinScheduleRange(result.ExaminationDate);
                warnings.AddRange(age.Messenger.Warnings);
                break;
            case "decline":
            case "location":
            case "examiner":
            case "deviation":
            case "bmi-warning":
                var interaction = new ExtractedInteractions {
                    KdpExaminationResultTo = result, ExaminationDate = result.ExaminationDate,
                    locationList = new() { new("SYNTH", "Y") },
                    examinerList = new() { new("SYNTH", "Y"), new("UNUSED", "N") },
                    deviationRanges = new(() => new() { new("Height", 40, 60) }),
                    prescribedRanges = new(() => new() { new("BMI", 10, 20) })
                };
                if (input["nullLookup"]?.GetValue<bool>() == true) { interaction.locationList = null; interaction.examinerList = null; }
                switch (scenario["when"].GetValue<string>()) {
                    case "decline":
                        var measurement = input["measurement"].GetValue<string>();
                        var declined = input["declined"].GetValue<bool>();
                        typeof(ExtractedInteractions).GetProperty("Is" + measurement + "Declined").SetValue(interaction, declined);
                        observed["value"] = JsonSerializer.SerializeToNode(typeof(ExtractedInteractions).GetProperty(measurement).GetValue(interaction));
                        observed["flag"] = measurement switch { "Height" => result.HeightMeasurementDeclined, "Weight" => result.WeightMeasurementDeclined, _ => result.HeadMeasurementDeclined };
                        break;
                    case "location": observed["valid"] = interaction.ValidateLocationCode(input["code"].GetValue<string>()); break;
                    case "examiner": observed["valid"] = interaction.ValidateExaminerCode(input["code"].GetValue<string>()); break;
                    case "deviation": interaction.CheckDeviations(result.Height, LookupsUtility.ChildMeasurementType.Height); break;
                    case "bmi-warning": interaction.CheckBmi(interaction, new System.ComponentModel.PropertyChangedEventArgs("Bmi")); break;
                }
                errors.AddRange(interaction.Messenger.Errors);
                warnings.AddRange(interaction.Messenger.Warnings);
                break;
            case "bmi": observed["bmi"] = new ExtractedBmi().CalculateBmi(result.Height.Value, result.Weight.Value); break;
            default: throw new InvalidOperationException("Unknown scenario operation");
        }
    }
    catch (ValidationWarningFault warning) { warnings.Add(warning.Message); }
    catch (ValidationFault error)
    {
        errors.AddRange(error.Message.Split('\n', StringSplitOptions.RemoveEmptyEntries)
            .Select(line => System.Text.RegularExpressions.Regex.Replace(line.Trim(), @"^\d+: ", "")));
    }
    observed["errors"] = JsonSerializer.SerializeToNode(errors);
    observed["warnings"] = JsonSerializer.SerializeToNode(warnings);
    observations.Add(new JsonObject { ["id"] = scenario["id"].DeepClone(), ["observation"] = observed });
}
File.WriteAllText(args[2], observations.ToJsonString(new() { WriteIndented = true }));
