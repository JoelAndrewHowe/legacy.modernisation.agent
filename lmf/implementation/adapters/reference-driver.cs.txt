using System.Text.Json;
using System.Text.Json.Nodes;
using Orvex.Business.Validation;
using Orvex.Messages.Faults;
using Orvex.TransferObjects;

var options = new JsonSerializerOptions { PropertyNameCaseInsensitive = true };
var scenarios = JsonNode.Parse(File.ReadAllText(args[0])).AsArray();
var fixtures = JsonNode.Parse(File.ReadAllText(args[1]));
ReferenceClock.Today = DateTime.Parse(fixtures["today"].GetValue<string>());
var birth = DateTime.Parse(fixtures["dateOfBirth"].GetValue<string>());
var observations = new JsonArray();
foreach (var scenario in scenarios)
{
    var input = scenario["input"];
    var result = input.Deserialize<KdpExaminationResultTO>(options);
    var errors = new List<string>();
    var warnings = new List<string>();
    var observed = new JsonObject();
    var validator = new KdpExaminationValidator();
    try
    {
        switch (scenario["when"].GetValue<string>())
        {
            case "lookup-validation-lifecycle":
                var validationInteraction = new ExtractedInteractions { KdpExaminationResultTo = result };
                var validationKind = input["kind"].GetValue<string>();
                var validationMode = input["mode"].GetValue<string>();
                if (validationMode == "missing-cached") {
                    validationInteraction.locationList = validationInteraction.lookupsService.Rows;
                    validationInteraction.examinerList = validationInteraction.lookupsService.Rows;
                }
                if (validationMode == "empty-rows") validationInteraction.lookupsService.Rows = new();
                validationInteraction.lookupsService.Fail = validationMode == "fault-recovery";
                var validationCode = input["code"]?.GetValue<string>();
                var firstValid = validationKind == "location" ? validationInteraction.ValidateLocationCode(validationCode) : validationInteraction.ValidateExaminerCode(validationCode);
                validationInteraction.lookupsService.Fail = false;
                var secondValid = validationKind == "location" ? validationInteraction.ValidateLocationCode(validationCode) : validationInteraction.ValidateExaminerCode(validationCode);
                observed["valid"] = JsonSerializer.SerializeToNode(new[] { firstValid, secondValid });
                observed["requests"] = JsonSerializer.SerializeToNode(validationInteraction.lookupsService.Requests);
                observed["prompts"] = JsonSerializer.SerializeToNode(validationInteraction.Messenger.ErrorPrompts);
                observed["cached"] = (validationKind == "location" ? validationInteraction.locationList : validationInteraction.examinerList) != null;
                errors.AddRange(validationInteraction.Messenger.Errors);
                break;
            case "lookup-retrieval":
                var lookupInteraction = new ExtractedInteractions { KdpExaminationResultTo = result };
                var lookupKind = input["kind"].GetValue<string>();
                var lookupMode = input["mode"].GetValue<string>();
                if (lookupMode == "cached" || lookupMode == "missing") {
                    lookupInteraction.locationList = lookupInteraction.lookupsService.Rows;
                    lookupInteraction.examinerList = lookupInteraction.lookupsService.Rows;
                }
                lookupInteraction.lookupsService.Fail = lookupMode == "fault";
                var firstLookup = lookupKind == "location" ? lookupInteraction.GetLocationList() : lookupInteraction.GetUsedExaminerList();
                var secondLookup = lookupKind == "location" ? lookupInteraction.GetLocationList() : lookupInteraction.GetUsedExaminerList();
                observed["codes"] = secondLookup == null ? null : JsonSerializer.SerializeToNode(secondLookup.Select(row => row.Code));
                observed["sameRows"] = firstLookup != null && secondLookup != null && firstLookup.Count == secondLookup.Count && firstLookup.Zip(secondLookup).All(pair => ReferenceEquals(pair.First, pair.Second));
                observed["requests"] = JsonSerializer.SerializeToNode(lookupInteraction.lookupsService.Requests);
                observed["prompts"] = JsonSerializer.SerializeToNode(lookupInteraction.Messenger.ErrorPrompts);
                errors.AddRange(lookupInteraction.Messenger.Errors);
                break;
            case "repeat-flags":
                var repeatUow = new RepeatUnitOfWork();
                foreach (var row in input["rows"].AsArray()) {
                    var exam = new ChildExamination {
                        PkChildExaminationId = row["id"].GetValue<long>(),
                        ClientId = row["client"].GetValue<long>(),
                        FkExaminationId = row["code"].GetValue<string>(),
                        IsRepeat = row["flag"].GetValue<string>()
                    };
                    repeatUow.Exams.Add(exam);
                    var given = new ChildExaminationGiven {
                        PkFkChildExamGivenId = exam.PkChildExaminationId,
                        ExaminationDate = DateTime.Parse(row["date"].GetValue<string>()),
                        ChildExamination = input["detached"].GetValue<bool>() ? null : exam
                    };
                    repeatUow.Given.Add(given);
                    repeatUow.Given.DatabaseRows.Add(new() {
                        PkFkChildExamGivenId = exam.PkChildExaminationId,
                        ExaminationDate = given.ExaminationDate, ChildExamination = exam
                    });
                }
                ExtractedRepeatFlags.UpdateRepeatFlags(repeatUow, result.ClientId, input["requestedCode"].GetValue<string>());
                var firstFlags = repeatUow.Exams.Select(exam => exam.IsRepeat).ToArray();
                observed["flags"] = JsonSerializer.SerializeToNode(firstFlags);
                observed["navigationRestored"] = repeatUow.Given.All(given => ReferenceEquals(given.ChildExamination,
                    repeatUow.Exams.Single(exam => exam.PkChildExaminationId == given.PkFkChildExamGivenId)));
                ExtractedRepeatFlags.UpdateRepeatFlags(repeatUow, result.ClientId, input["requestedCode"].GetValue<string>());
                observed["idempotent"] = firstFlags.SequenceEqual(repeatUow.Exams.Select(exam => exam.IsRepeat));
                break;
            case "weeks-between":
                observed["weeks"] = birth.WeeksBetween(result.ExaminationDate);
                break;
            case "new-result":
                var creator = new ExtractedLifecycle();
                creator.ato.Id = fixtures["clientId"].ToString();
                creator.ato.HasPreschoolConsent = input["hasConsentFixture"].GetValue<bool>();
                creator.listOfKdpExaminationResultDelegates.Add(new() { KdpExaminationResultTo = new() { PkChildExaminationId = 42 } });
                var created = creator.AddNewKdpResult();
                var defaultProperties = new[] { "ClientId", "PkChildExaminationId", "ResultEnteredDate", "Origin", "Reschedule", "HeightMeasurementDeclined", "WeightMeasurementDeclined", "HeadMeasurementDeclined", "EnabledPropChangeEvents", "ExaminationCode", "ExaminationDate" };
                var defaults = new JsonObject();
                foreach (var property in defaultProperties)
                    defaults[property] = JsonSerializer.SerializeToNode(typeof(KdpExaminationResultTO).GetProperty(property).GetValue(created.KdpExaminationResultTo));
                observed["defaults"] = defaults;
                observed["context"] = new JsonObject {
                    ["gender"] = created.gender,
                    ["birth"] = created.dateOfBirth.ToString("yyyy-MM-dd"),
                    ["death"] = created.dateOfDeath?.ToString("yyyy-MM-dd"),
                    ["sameExisting"] = ReferenceEquals(creator.ato.HcpExaminationResultToList, created.existingResults),
                    ["scheduled"] = created.IsScheduled, ["unscheduled"] = created.IsUnscheduled,
                    ["description"] = created.KdpExaminationResultTo.ExamDescription,
                    ["notifications"] = JsonSerializer.SerializeToNode(created.Notifications),
                    ["lazyRanges"] = !created.deviationRanges.IsValueCreated && !created.recommendedAgeRangeInWeeks.IsValueCreated,
                    ["requests"] = created.measurementsService.RangeRequests.Count + created.KdpExaminationService.Requests.Count
                };
                observed["insertedFirst"] = ReferenceEquals(creator.listOfKdpExaminationResultDelegates[0], created);
                observed["count"] = creator.listOfKdpExaminationResultDelegates.Count;
                observed["confirmations"] = JsonSerializer.SerializeToNode(creator.messenger.Confirmations);
                warnings.AddRange(creator.messenger.Warnings);
                break;
            case "ui-insert":
                var insertControl = new ExtractedSaveControl { setDefaultTabPage = false };
                insertControl.kdpExaminationDelegate.ato.Id = fixtures["clientId"].ToString();
                insertControl.kdpExaminationDelegate.ato.HasPreschoolConsent = input["hasConsentFixture"].GetValue<bool>();
                insertControl.kdpExaminationDelegate.listOfKdpExaminationResultDelegates.Add(new() {
                    KdpExaminationResultTo = new() { PkChildExaminationId = 42 }
                });
                foreach (var key in input["keys"].AsArray())
                    insertControl.hlxAdminTabControlKdpExamination.TabPages.Add(new(key.GetValue<string>()));
                insertControl.hlxAdminTabControlKdpExamination.SelectedTab = new("42");
                insertControl.toolStripButtonInsert_Click(insertControl, EventArgs.Empty);
                // Repeated click must use the newly inserted first tab as the guard.
                insertControl.setDefaultTabPage = false;
                insertControl.toolStripButtonInsert_Click(insertControl, EventArgs.Empty);
                observed["keys"] = JsonSerializer.SerializeToNode(insertControl.hlxAdminTabControlKdpExamination.TabPages.Select(tab => tab.Key));
                observed["heading"] = insertControl.hlxAdminTabControlKdpExamination.TabPages[0].Text;
                observed["defaultTab"] = insertControl.setDefaultTabPage;
                var insertResults = insertControl.kdpExaminationDelegate.listOfKdpExaminationResultDelegates;
                observed["resultIds"] = JsonSerializer.SerializeToNode(insertResults.Select(item => item.KdpExaminationResultTo.PkChildExaminationId));
                observed["selected"] = insertControl.hlxAdminTabControlKdpExamination.SelectedTab.Key;
                observed["resultsAtSetup"] = JsonSerializer.SerializeToNode(insertControl.ResultsAtSetup);
                observed["keysAtSetup"] = JsonSerializer.SerializeToNode(insertControl.KeysAtSetup);
                observed["confirmations"] = JsonSerializer.SerializeToNode(insertControl.kdpExaminationDelegate.messenger.Confirmations);
                observed["defaults"] = insertResults.Count > 1 ? new JsonObject {
                    ["ClientId"] = insertResults[0].KdpExaminationResultTo.ClientId,
                    ["Origin"] = insertResults[0].KdpExaminationResultTo.Origin,
                    ["Reschedule"] = insertResults[0].KdpExaminationResultTo.Reschedule
                } : null;
                warnings.AddRange(insertControl.kdpExaminationDelegate.messenger.Warnings);
                break;
            case "tab-change":
                var tabControl = new ExtractedSaveControl { processingTabs = input["processing"].GetValue<bool>() };
                var selectedKey = input["selected"]?.GetValue<string>();
                tabControl.hlxAdminTabControlKdpExamination.SelectedTab = selectedKey == null ? null : new(selectedKey);
                var tabEdited = new KdpExaminationResultTO { PkChildExaminationId = 42, Height = 99, ChangedFixture = true };
                var tabWrapped = new KdpExaminationResultDelegate { KdpExaminationResultTo = tabEdited };
                tabControl.kdpExaminationDelegate.listOfKdpExaminationResultDelegates.Add(tabWrapped);
                tabControl.kdpExaminationDelegate.service.Stored = new() { PkChildExaminationId = 42, Height = 55 };
                tabControl.kdpExaminationDelegate.messenger.ConfirmationResponse = input["response"].GetValue<bool>();
                var pageKey = input["page"]?.GetValue<string>();
                var tabEvent = new TabCancelEventArgs { Page = pageKey == null ? null : new(pageKey), Cancel = input["initialCancel"].GetValue<bool>() };
                tabControl.hlxAdminTabControlKdpExamination_ChangingSelectedTab(tabControl, tabEvent);
                observed["cancelled"] = tabEvent.Cancel;
                observed["confirmations"] = JsonSerializer.SerializeToNode(tabControl.kdpExaminationDelegate.messenger.Confirmations);
                observed["reloadCalls"] = JsonSerializer.SerializeToNode(tabControl.kdpExaminationDelegate.service.Calls);
                observed["height"] = JsonSerializer.SerializeToNode(tabWrapped.KdpExaminationResultTo.Height);
                observed["prepared"] = JsonSerializer.SerializeToNode(tabControl.PreparedTabs);
                observed["reloadsAtPreparation"] = JsonSerializer.SerializeToNode(tabControl.ReloadsAtPreparation);
                observed["selected"] = tabControl.hlxAdminTabControlKdpExamination.SelectedTab?.Key;
                break;
            case "ui-save":
                var saveControl = new ExtractedSaveControl { Editable = false, currentTab = new("99") };
                var uiOutcome = input["outcome"].GetValue<string>();
                saveControl.hlxAdminTabControlKdpExamination.SelectedTab = uiOutcome == "no-tab" ? null : new("42");
                var uiSelected = new KdpExaminationResultTO { PkChildExaminationId = 42 };
                saveControl.kdpExaminationDelegate.listOfKdpExaminationResultDelegates.Add(new() {
                    KdpExaminationResultTo = new() { PkChildExaminationId = 99 }
                });
                saveControl.kdpExaminationDelegate.listOfKdpExaminationResultDelegates.Add(new() { KdpExaminationResultTo = uiSelected });
                saveControl.kdpExaminationDelegate.service.SaveOutcome = uiOutcome;
                observed["allowed"] = saveControl.SaveChanges();
                var uiCalls = saveControl.kdpExaminationDelegate.service.SaveCalls;
                observed["calls"] = uiCalls.Count;
                observed["sameContext"] = uiCalls.Count == 1 && ReferenceEquals(uiCalls[0].Context, saveControl.kdpExaminationDelegate.ato);
                observed["sameResult"] = uiCalls.Count == 1 && ReferenceEquals(uiCalls[0].Result, uiSelected);
                observed["currentIsSelected"] = ReferenceEquals(saveControl.currentTab, saveControl.hlxAdminTabControlKdpExamination.SelectedTab);
                observed["defaultTab"] = saveControl.setDefaultTabPage;
                observed["refreshes"] = JsonSerializer.SerializeToNode(saveControl.Refreshes);
                observed["prompts"] = JsonSerializer.SerializeToNode(saveControl.kdpExaminationDelegate.messenger.ErrorPrompts);
                errors.AddRange(saveControl.kdpExaminationDelegate.messenger.Errors);
                break;
            case "parent-save":
                var saver = new ExtractedLifecycle();
                var selected = new KdpExaminationResultTO { PkChildExaminationId = 42, Height = 55 };
                var unrelated = new KdpExaminationResultTO { PkChildExaminationId = 99, Height = 66 };
                saver.listOfKdpExaminationResultDelegates.Add(new() { KdpExaminationResultTo = unrelated });
                saver.listOfKdpExaminationResultDelegates.Add(new() { KdpExaminationResultTo = selected });
                saver.service.SaveOutcome = input["outcome"].GetValue<string>();
                observed["allowed"] = saver.Save(42);
                observed["calls"] = saver.service.SaveCalls.Count;
                observed["sameContext"] = saver.service.SaveCalls.All(call => ReferenceEquals(call.Context, saver.ato));
                observed["sameResult"] = saver.service.SaveCalls.All(call => ReferenceEquals(call.Result, selected));
                observed["unrelatedUnchanged"] = unrelated.Height == 66 && unrelated.PkChildExaminationId == 99;
                observed["prompts"] = JsonSerializer.SerializeToNode(saver.messenger.ErrorPrompts);
                errors.AddRange(saver.messenger.Errors);
                break;
            case "cancel-changes":
                var lifecycle = new ExtractedLifecycle();
                var edited = new KdpExaminationResultTO { PkChildExaminationId = input["id"].GetValue<long>(), Height = 99, ChangedFixture = input["changed"].GetValue<bool>() };
                lifecycle.service.Stored = new() { PkChildExaminationId = 42, Height = 55 };
                var wrapped = new KdpExaminationResultDelegate { KdpExaminationResultTo = edited };
                if (!input["empty"].GetValue<bool>()) lifecycle.listOfKdpExaminationResultDelegates.Add(wrapped);
                lifecycle.messenger.ConfirmationResponse = input["response"].GetValue<bool>();
                observed["allowed"] = lifecycle.ConfirmCancelChanges(edited.PkChildExaminationId);
                observed["reloadCalls"] = JsonSerializer.SerializeToNode(lifecycle.service.Calls);
                observed["height"] = input["empty"].GetValue<bool>() ? null : JsonSerializer.SerializeToNode(wrapped.KdpExaminationResultTo.Height);
                observed["replaced"] = !ReferenceEquals(edited, wrapped.KdpExaminationResultTo);
                observed["confirmations"] = JsonSerializer.SerializeToNode(lifecycle.messenger.Confirmations);
                break;
            case "code-control":
                var codeDelegate = new ExtractedInteractions { KdpExaminationResultTo = result };
                codeDelegate.Messenger.LookupResponse = input["lookup"]?.GetValue<string>();
                codeDelegate.Messenger.ConfirmationResponse = input["response"].GetValue<bool>();
                var codeControls = new ExtractedCodeControl { kdpExaminationResultDelegate = codeDelegate };
                codeControls.hlxTextBoxExamCode.Text = input["text"].GetValue<string>();
                var codeCancel = new System.ComponentModel.CancelEventArgs(input["initialCancel"].GetValue<bool>());
                if (input["button"].GetValue<bool>())
                    codeControls.hlxTextBoxExamCode_ButtonClick(codeControls.hlxTextBoxExamCode, EventArgs.Empty);
                else
                    codeControls.hlxTextBoxExamCode_Validating(codeControls.hlxTextBoxExamCode, codeCancel);
                observed["selected"] = result.ExaminationCode;
                observed["cancelled"] = codeCancel.Cancel;
                observed["confirmations"] = JsonSerializer.SerializeToNode(codeDelegate.Messenger.Confirmations);
                observed["lookupRequests"] = codeDelegate.Messenger.LookupRequests.Count;
                observed["lookupCodes"] = JsonSerializer.SerializeToNode(codeDelegate.Messenger.LookupRequests);
                errors.AddRange(codeDelegate.Messenger.Errors);
                warnings.AddRange(codeDelegate.Messenger.Warnings);
                break;
            case "dropdown-control":
                var dropdownDelegate = new ExtractedInteractions { KdpExaminationResultTo = result };
                var dropdownControls = new ExtractedDropdownControls { kdpExaminationResultDelegate = dropdownDelegate };
                var dropdown = new HlxDropDown { Text = input["descriptor"].GetValue<string>() };
                var dropdownCancel = new System.ComponentModel.CancelEventArgs(input["initialCancel"].GetValue<bool>());
                if (input["cancelOverload"].GetValue<bool>())
                    dropdownControls.ValidateExaminerTypeInput(dropdown, dropdownCancel);
                else {
                    switch (input["kind"].GetValue<string>()) {
                        case "LOCATON_TYPE": dropdownControls.ValidateLocationTypeInput(dropdown, EventArgs.Empty); break;
                        case "EXAMINER_TYPE": dropdownControls.ValidateExaminerTypeInput(dropdown, EventArgs.Empty); break;
                        case "REASON_NOT_EXAMINED": dropdownControls.ValidateReasonNotExaminedTypeInput(dropdown, EventArgs.Empty); break;
                        default: throw new InvalidOperationException("Unknown dropdown kind");
                    }
                }
                observed["text"] = dropdown.Text;
                observed["cancelled"] = dropdownCancel.Cancel;
                errors.AddRange(dropdownDelegate.Messenger.Errors);
                warnings.AddRange(dropdownDelegate.Messenger.Warnings);
                break;
            case "date-control":
                var dateDelegate = new ExtractedInteractions {
                    KdpExaminationResultTo = result,
                    dateOfBirth = birth,
                    dateOfDeath = input["deathFixture"] == null ? null : DateTime.Parse(input["deathFixture"].GetValue<string>()),
                    recommendedAgeRangeInWeeks = new(() => new(6, 8))
                };
                if (input["duplicateFixture"].GetValue<bool>())
                    dateDelegate.existingResults.Add(new() { PkChildExaminationId = 42, ExaminationCode = "KD03", ExaminationDate = result.ExaminationDate });
                var dateControls = new ExtractedDateControl { kdpExaminationResultDelegate = dateDelegate };
                dateControls.hlxDateBoxExamDate.Text = input["text"].GetValue<string>();
                dateControls.hlxDateBoxExamDate.Value = DateTime.Parse(input["value"].GetValue<string>());
                var dateCancel = new System.ComponentModel.CancelEventArgs(input["initialCancel"].GetValue<bool>());
                dateControls.hlxDateBoxExamDate_Validating(dateControls.hlxDateBoxExamDate, dateCancel);
                observed["cancelled"] = dateCancel.Cancel;
                errors.AddRange(dateDelegate.Messenger.Errors);
                warnings.AddRange(dateDelegate.Messenger.Warnings);
                break;
            case "measurement-control":
                var controlDelegate = new ExtractedInteractions {
                    KdpExaminationResultTo = result,
                    deviationRanges = new(() => new() { new("Height", 40, 60), new("Weight", 3, 6) })
                };
                var controls = new ExtractedMeasurementControls { kdpExaminationResultDelegate = controlDelegate };
                var field = input["field"].GetValue<string>();
                var control = (ControlFixture)typeof(ExtractedMeasurementControls).GetField("hlxNumericEditBoxCustom" + field).GetValue(controls);
                control.Text = input["text"].GetValue<string>();
                control.Value = input["value"].GetValue<decimal>();
                var cancel = new System.ComponentModel.CancelEventArgs(input["initialCancel"].GetValue<bool>());
                typeof(ExtractedMeasurementControls).GetMethod("hlxNumericEditBoxCustom" + field + "_Validating").Invoke(controls, new object[] { control, cancel });
                observed["cancelled"] = cancel.Cancel;
                errors.AddRange(controlDelegate.Messenger.Errors);
                warnings.AddRange(controlDelegate.Messenger.Warnings);
                break;
            case "common-recording":
                var common = new ExtractedInteractions { KdpExaminationResultTo = result };
                var commonField = input["field"].GetValue<string>();
                var commonProperty = typeof(ExtractedInteractions).GetProperty(commonField);
                var commonValue = input["value"].Deserialize(commonProperty.PropertyType);
                commonProperty.SetValue(common, commonValue);
                commonProperty.SetValue(common, commonValue); // unchanged assignment
                observed["assigned"] = JsonSerializer.SerializeToNode(commonProperty.GetValue(common));
                if (input["clear"].GetValue<bool>())
                    commonProperty.SetValue(common, commonField == "WithdrawConsent" ? (object)false : null);
                observed["displayed"] = JsonSerializer.SerializeToNode(commonProperty.GetValue(common));
                var commonValues = new JsonObject();
                foreach (var name in new[] { "ReasonNotExamined", "Consent", "CoreComponent", "Reschedule", "WithdrawConsent", "Origin" })
                    commonValues[name] = JsonSerializer.SerializeToNode(typeof(KdpExaminationResultTO).GetProperty(name).GetValue(result));
                observed["stored"] = commonValues;
                observed["notifications"] = JsonSerializer.SerializeToNode(common.Notifications);
                break;
            case "physical-assessment":
                var physical = new ExtractedInteractions { KdpExaminationResultTo = result };
                var physicalField = input["field"].GetValue<string>();
                var physicalProperty = typeof(ExtractedInteractions).GetProperty(physicalField);
                physicalProperty.SetValue(physical, (byte?)2);
                physicalProperty.SetValue(physical, (byte?)2); // unchanged assignment
                observed["assigned"] = JsonSerializer.SerializeToNode(physicalProperty.GetValue(physical));
                if (input["clear"].GetValue<bool>()) physicalProperty.SetValue(physical, null);
                observed["displayed"] = JsonSerializer.SerializeToNode(physicalProperty.GetValue(physical));
                var physicalValues = new JsonObject();
                foreach (var name in new[] { "Heart", "Hips", "Eyes", "Testes", "PhysicalGeneral" })
                    physicalValues[name] = JsonSerializer.SerializeToNode(typeof(KdpExaminationResultTO).GetProperty(name).GetValue(result));
                observed["stored"] = physicalValues;
                observed["notifications"] = JsonSerializer.SerializeToNode(physical.Notifications);
                observed["accepted"] = false;
                validator.ValidateOnSave(result, birth, null, new());
                observed["accepted"] = true; // validator only, not persistence
                break;
            case "save":
                validator.ValidateOnSave(result, birth, null, new());
                // Detached observation only, not a claim of database read-back.
                observed["saved"] = input.DeepClone();
                break;
            case "head-field": validator.ValidateHeadCircumference(result.HeadCircumference.Value); break;
            case "duplicate-date":
                validator.ValidateExamCodeForDate(new() { new() { ExaminationCode = "KD03", ExaminationDate = result.ExaminationDate, PkChildExaminationId = 42 } }, 0, "KD03", result.ExaminationDate);
                break;
            case "duplicate-code":
                validator.ValidateExamCodeForDuplicates(new() { new() { ExaminationCode = "KD03", PkChildExaminationId = 42 } }, "KD03", 0);
                break;
            case "consent": validator.ValidateKdpConsent(result.ClientId, false); break;
            case "age":
                var age = new ExtractedAge { dateOfBirth = birth, recommendedAgeRangeInWeeks = new(() => new(fixtures["scheduleWeeks"][0].GetValue<double>(), fixtures["scheduleWeeks"][1].GetValue<double>())) };
                age.CheckAgeWithinScheduleRange(result.ExaminationDate);
                warnings.AddRange(age.Messenger.Warnings);
                break;
            case "populate":
            case "code-setter":
            case "date-setter":
            case "measurement-setter":
            case "measurement-assignment":
            case "code-selection":
            case "code-selection-sequence":
            case "decline":
            case "location":
            case "examiner-category":
            case "examiner-code":
            case "location-category":
            case "location-code":
            case "examiner":
            case "numeric-descriptor":
            case "description-accessor":
            case "deviation":
            case "deviation-guard":
            case "bmi-event":
            case "bmi-warning":
                var interaction = new ExtractedInteractions {
                    KdpExaminationResultTo = result,
                    locationList = new() { new("SYNTH", "Y") },
                    examinerList = new() { new("SYNTH", "Y"), new("UNUSED", "N") },
                    deviationRanges = new(() => new() { new("Height", 40, 60) }),
                    prescribedRanges = new(() => new() { new("BMI", 10, 20) })
                };
                if (input["nullLookup"]?.GetValue<bool>() == true) { interaction.locationList = null; interaction.examinerList = null; interaction.lookupsService.Rows = null; }
                switch (scenario["when"].GetValue<string>()) {
                    case "populate":
                        // Start with a distinct sentinel so a missing DTO assignment cannot pass.
                        interaction.KdpExaminationResultTo = new() {
                            ExaminationCode = "KD08", ExaminationDate = new DateTime(2000, 1, 1)
                        };
                        var populateBirth = new DateTime(2024, 12, 1);
                        DateTime? death = input["withDeath"].GetValue<bool>() ? new DateTime(2025, 2, 1) : null;
                        var populateExisting = new List<KdpExaminationResultTO> { new() { PkChildExaminationId = 42 } };
                        var oldDeviation = interaction.deviationRanges;
                        interaction.recommendedAgeRangeInWeeks = new(() => new(0, 0));
                        var oldSchedule = interaction.recommendedAgeRangeInWeeks;
                        for (var repeat = 0; repeat < 2; repeat++) {
                            if (input["scheduled"].GetValue<bool>())
                                interaction.Populate(result, "F", populateBirth, death, populateExisting, true);
                            else
                                interaction.Populate(result, "F", populateBirth, death, populateExisting);
                        }
                        observed["sameDto"] = ReferenceEquals(result, interaction.KdpExaminationResultTo);
                        observed["sameExisting"] = ReferenceEquals(populateExisting, interaction.existingResults);
                        observed["gender"] = interaction.gender;
                        observed["birth"] = interaction.dateOfBirth.ToString("yyyy-MM-dd");
                        observed["death"] = interaction.dateOfDeath?.ToString("yyyy-MM-dd");
                        observed["scheduled"] = interaction.IsScheduled;
                        observed["unscheduled"] = interaction.IsUnscheduled;
                        observed["description"] = interaction.KdpExaminationResultTo.ExamDescription;
                        observed["selected"] = interaction.ExaminationCode;
                        observed["date"] = interaction.ExaminationDate.ToString("yyyy-MM-ddTHH:mm:ss");
                        observed["rangesReset"] = !ReferenceEquals(oldDeviation, interaction.deviationRanges) && !ReferenceEquals(oldSchedule, interaction.recommendedAgeRangeInWeeks);
                        observed["lazyBeforeRead"] = !interaction.deviationRanges.IsValueCreated && !interaction.recommendedAgeRangeInWeeks.IsValueCreated;
                        observed["requestsBeforeRead"] = interaction.measurementsService.RangeRequests.Count + interaction.KdpExaminationService.Requests.Count;
                        _ = interaction.deviationRanges.Value;
                        _ = interaction.recommendedAgeRangeInWeeks.Value;
                        observed["rangeRequests"] = JsonSerializer.SerializeToNode(interaction.measurementsService.RangeRequests);
                        observed["scheduleRequests"] = JsonSerializer.SerializeToNode(interaction.KdpExaminationService.Requests);
                        observed["lookupDescriptions"] = JsonSerializer.SerializeToNode(interaction.LocationTypeList.Concat(interaction.ExaminerTypeList).Concat(interaction.NotExaminedReasonList).Select(row => row.Description));
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "code-setter":
                        interaction.recommendedAgeRangeInWeeks = new(() => interaction.KdpExaminationService.GetAgeRangeFromExamSchedule(interaction.ExaminationCode));
                        var priorSchedule = interaction.recommendedAgeRangeInWeeks;
                        interaction.ExamCodeValidated = input["validated"].GetValue<bool>();
                        interaction.ExaminationCode = input["code"].GetValue<string>();
                        observed["selected"] = result.ExaminationCode;
                        observed["heart"] = result.Heart;
                        observed["development"] = result.Development;
                        observed["dsk"] = result.DskLocomotor;
                        observed["rsk"] = result.RskEngagement;
                        observed["description"] = result.ExamDescription;
                        observed["rangesReset"] = !ReferenceEquals(priorSchedule, interaction.recommendedAgeRangeInWeeks);
                        observed["lazyBeforeRead"] = !interaction.recommendedAgeRangeInWeeks.IsValueCreated;
                        _ = interaction.recommendedAgeRangeInWeeks.Value;
                        observed["scheduleRequests"] = JsonSerializer.SerializeToNode(interaction.KdpExaminationService.Requests);
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "date-setter":
                        var priorRanges = interaction.deviationRanges;
                        interaction.ExaminationDate = DateTime.Parse(input["newDate"].GetValue<string>());
                        observed["date"] = interaction.ExaminationDate.ToString("yyyy-MM-ddTHH:mm:ss");
                        observed["rangesReset"] = !ReferenceEquals(priorRanges, interaction.deviationRanges);
                        observed["lazyBeforeRead"] = !interaction.deviationRanges.IsValueCreated;
                        _ = interaction.deviationRanges.Value;
                        observed["rangeRequests"] = JsonSerializer.SerializeToNode(interaction.measurementsService.RangeRequests);
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "measurement-setter":
                        interaction.Weight = input["newWeight"].GetValue<decimal>();
                        observed["weight"] = result.Weight;
                        observed["bmi"] = interaction.Bmi;
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "measurement-assignment":
                        var assignedField = input["field"].GetValue<string>();
                        var assignedValue = input["value"]?.GetValue<decimal>();
                        for (var assignment = 0; assignment < 2; assignment++)
                            typeof(ExtractedInteractions).GetProperty(assignedField).SetValue(interaction, assignedValue);
                        observed["value"] = JsonSerializer.SerializeToNode(typeof(ExtractedInteractions).GetProperty(assignedField).GetValue(interaction));
                        observed["bmi"] = interaction.Bmi;
                        observed["outside"] = interaction.IsBmiOutsideRange;
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "code-selection":
                    case "code-selection-sequence":
                        interaction.Messenger.ConfirmationResponse = input["response"]?.GetValue<bool>() ?? false;
                        if (input["responses"] is JsonArray responses)
                            foreach (var response in responses)
                                interaction.Messenger.ConfirmationResponses.Enqueue(response.GetValue<bool>());
                        if (input["duplicate"]?.GetValue<bool>() == true)
                            interaction.existingResults.Add(new() { ExaminationCode = "KD03", PkChildExaminationId = 42, ExaminationDate = result.ExaminationDate.AddDays(-1) });
                        if (input["existingResult"] is JsonObject existingResult)
                        {
                            result.PkChildExaminationId = input["resultId"].GetValue<long>();
                            interaction.existingResults.Add(new() {
                                PkChildExaminationId = existingResult["id"].GetValue<long>(),
                                ExaminationCode = existingResult["code"].GetValue<string>(),
                                ExaminationDate = DateTime.Parse(existingResult["date"].GetValue<string>()) });
                        }
                        // A previously valid selection must become invalid on either rejection.
                        interaction.ExamCodeValidated = true;
                        observed["valid"] = interaction.ValidateAndSetExamCode(input["code"]?.GetValue<string>() ?? "KD03");
                        if (scenario["when"].GetValue<string>() == "code-selection-sequence") {
                            observed["validated"] = interaction.ExamCodeValidated;
                            observed["remainingResponses"] = interaction.Messenger.ConfirmationResponses.Count;
                        }
                        observed["selected"] = result.ExaminationCode;
                        observed["confirmations"] = JsonSerializer.SerializeToNode(interaction.Messenger.Confirmations);
                        break;
                    case "decline":
                        var measurement = input["measurement"].GetValue<string>();
                        var declined = input["declined"].GetValue<bool>();
                        typeof(ExtractedInteractions).GetProperty("Is" + measurement + "Declined").SetValue(interaction, declined);
                        observed["value"] = JsonSerializer.SerializeToNode(typeof(ExtractedInteractions).GetProperty(measurement).GetValue(interaction));
                        observed["flag"] = measurement switch { "Height" => result.HeightMeasurementDeclined, "Weight" => result.WeightMeasurementDeclined, _ => result.HeadMeasurementDeclined };
                        observed["required"] = JsonSerializer.SerializeToNode(typeof(ExtractedInteractions).GetProperty("Is" + measurement + "Required").GetValue(interaction));
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "description-accessor":
                        interaction.ExamDescription = input["description"]?.GetValue<string>();
                        observed["displayed"] = interaction.ExamDescription;
                        observed["stored"] = result.ExamDescription;
                        break;
                    case "numeric-descriptor":
                        observed["valid"] = interaction.ValidateNumericDescriptorEntered(input["descriptor"].GetValue<string>(), Enum.Parse<ExtractedInteractions.TYPE_OF_INPUT>(input["kind"].GetValue<string>()));
                        break;
                    case "examiner-category":
                        interaction.ExaminerCategory = input["category"]?.GetValue<string>();
                        observed["category"] = result.ExaminerCategory;
                        observed["type"] = result.ExaminerType;
                        observed["code"] = result.ExaminerHcpCode;
                        observed["name"] = result.ExaminerName;
                        observed["cacheCleared"] = interaction.examinerList == null;
                        interaction.lookupsService.Rows = new() { new("NEW", "Y", "Synthetic refreshed HCP", "NEW-TYPE") };
                        observed["refreshedCodes"] = JsonSerializer.SerializeToNode(interaction.GetUsedExaminerList()?.Select(row => row.Code));
                        observed["requests"] = JsonSerializer.SerializeToNode(interaction.lookupsService.Requests);
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "examiner-code":
                        interaction.examinerList = new() {
                            new("SYNTH", "N", "Unused same-code HCP", "WRONG-TYPE"),
                            new("SYNTH", "Y", "Synthetic HCP", "SYNTH-TYPE"),
                            new("UNUSED", "N", "Unused HCP", "UNUSED-TYPE") };
                        interaction.ExaminerKdpCode = input["code"]?.GetValue<string>();
                        observed["code"] = result.ExaminerHcpCode;
                        observed["category"] = result.ExaminerCategory;
                        observed["type"] = result.ExaminerType;
                        observed["name"] = result.ExaminerName;
                        observed["cacheCleared"] = interaction.examinerList == null;
                        observed["requests"] = JsonSerializer.SerializeToNode(interaction.lookupsService.Requests);
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "location-category":
                        interaction.LocationType = input["category"]?.GetValue<byte>();
                        observed["category"] = result.LocationType;
                        observed["code"] = result.LocationCode;
                        observed["name"] = result.LocationName;
                        observed["cacheCleared"] = interaction.locationList == null;
                        interaction.lookupsService.Rows = new() { new("NEW", "Y", "Synthetic refreshed location") };
                        observed["refreshedCodes"] = JsonSerializer.SerializeToNode(interaction.GetLocationList()?.Select(row => row.Code));
                        observed["requests"] = JsonSerializer.SerializeToNode(interaction.lookupsService.Requests);
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "location-code":
                        interaction.locationList = input["noCache"].GetValue<bool>() ? null : new() { new("SYNTH", "Y", "Synthetic location description") };
                        var originalLocationCache = interaction.locationList;
                        interaction.LocationCode = input["code"]?.GetValue<string>();
                        observed["code"] = result.LocationCode;
                        observed["name"] = result.LocationName;
                        observed["sameCache"] = ReferenceEquals(originalLocationCache, interaction.locationList);
                        observed["requests"] = JsonSerializer.SerializeToNode(interaction.lookupsService.Requests);
                        observed["notifications"] = JsonSerializer.SerializeToNode(interaction.Notifications);
                        break;
                    case "location": observed["valid"] = interaction.ValidateLocationCode(input["code"].GetValue<string>()); break;
                    case "examiner": observed["valid"] = interaction.ValidateExaminerCode(input["code"].GetValue<string>()); break;
                    case "deviation-guard":
                        interaction.deviationRanges = new(() => new() { new("Height", input["lower"].GetValue<double>(), input["upper"].GetValue<double>()) });
                        interaction.CheckDeviations(result.Height, LookupsUtility.ChildMeasurementType.Height);
                        observed["rangeRead"] = interaction.deviationRanges.IsValueCreated;
                        break;
                    case "bmi-event":
                        interaction.CheckBmi(interaction, new System.ComponentModel.PropertyChangedEventArgs(input["property"]?.GetValue<string>()));
                        observed["rangeRead"] = interaction.prescribedRanges.IsValueCreated;
                        break;
                    case "deviation": interaction.CheckDeviations(result.Height, LookupsUtility.ChildMeasurementType.Height); break;
                    case "bmi-warning": interaction.CheckBmi(interaction, new System.ComponentModel.PropertyChangedEventArgs("Bmi")); break;
                }
                errors.AddRange(interaction.Messenger.Errors);
                warnings.AddRange(interaction.Messenger.Warnings);
                break;
            case "bmi": observed["bmi"] = new ExtractedBmi().CalculateBmi(result.Height.Value, result.Weight.Value); break;
            default: throw new InvalidOperationException("Unknown scenario operation");
        }
    }
    catch (ValidationWarningFault warning) { warnings.Add(warning.Message); }
    catch (ValidationFault error)
    {
        errors.AddRange(error.Message.Split('\n', StringSplitOptions.RemoveEmptyEntries)
            .Select(line => System.Text.RegularExpressions.Regex.Replace(line.Trim(), @"^\d+: ", "")));
    }
    observed["errors"] = JsonSerializer.SerializeToNode(errors);
    observed["warnings"] = JsonSerializer.SerializeToNode(warnings);
    observations.Add(new JsonObject { ["id"] = scenario["id"].DeepClone(), ["observation"] = observed });
}
File.WriteAllText(args[2], observations.ToJsonString(new() { WriteIndented = true }));
