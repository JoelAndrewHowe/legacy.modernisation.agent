    [AspNetCompatibilityRequirements(RequirementsMode = AspNetCompatibilityRequirementsMode.Allowed)]
    public class PersonService : IPersonService
    {
        private readonly Func<IUnitOfWork> unitOfWorkFactory;
        private readonly IGlobalSessionValuesService sessionValuesService;
        public PersonService(Func<IUnitOfWork> unitOfWorkFactory, IGlobalSessionValuesService sessionValuesService)
        {
            this.unitOfWorkFactory = unitOfWorkFactory;
            this.sessionValuesService = sessionValuesService;
        }

        static PersonService()
        {
            Mapper.CreateMap<ChildGeneral, PersonTO>()
               .ForMember(dest => dest.Id, act => act.MapFrom(source => source.ClientId))
               .ForMember(dest => dest.Gender, act => act.MapFrom(source => source.Sex))
               .IgnoreUnMapped();
        }

        public PersonTO GetPersonByClientId(long clientId)
        {
            using (var uow = unitOfWorkFactory())
            {
                var child = uow.Repository<ChildGeneral>()
                            .AsNoTracking()
                            .Include(rec => rec.RemoteFlag)
                            .SingleOrDefault(c => c.ClientId == clientId);

                if (child == null)
                {
                    throw QueryNoRecordFault.Default("No client found with the requested Client ID");
                }

                var to = child.To<PersonTO>();

                to.Remote = child.RemoteFlag.IsRemote == OVXBoolean.True;

                to.AgeString = child.CurrentStatus == LookupsUtility.CurrentStatus_Stillbirth ? null : Lookups.Age(to.DateOfBirth, to.DateOfDeath, uow);

                var childCentres = uow.Repository<ChildCentres>()
                    .AsNoTracking()
                    .Include(cc => cc.ChildLocations)
                    .Where(cc => cc.ClientId == clientId)
                    .ToList();

                var wisCentres = uow.Repository<WisPatientCentre>()
                                     .AsNoTracking()
                                     .Include(wpc => wpc.Location)
                                     .Where(wpc => wpc.PatientRec.NhsNumber == to.NhsNumber
                                                        && wpc.Healthboard.Sectors.Any())
                                     .ToList();

                to.School = GetChildCentreText(childCentres.FirstOrDefault(cc => cc.ChildLocationType == (byte)LookupsUtility.CENTRE_TYPES.SCHOOL));
                to.ExamCentre = GetChildCentreText(childCentres.FirstOrDefault(cc => cc.ChildLocationType == (byte)LookupsUtility.CENTRE_TYPES.EXAM_CENTRE));
                to.TreatmentCentre = GetChildCentreText(childCentres.FirstOrDefault(cc => cc.ChildLocationType == (byte)LookupsUtility.CENTRE_TYPES.TREATMENT_CENTRE));
                to.Practice = GetChildCentreText(childCentres.FirstOrDefault(cc => cc.ChildLocationType == (byte)LookupsUtility.LOCATION_TYPES.PRACTICE));
                to.Organisation = GetOrganisationValue(childCentres, uow);
                to.SchoolImmLoc = GetPatientCentres(wisCentres.FirstOrDefault(cc => cc.FkCentreType == (byte)LookupsUtility.WIS_CENTRE_TYPES.SCHOOL_IMM_LOC));

                var hcps = uow.Repository<RelHcpDemographics>()
                    .AsNoTracking()
                    .Include(hcp => hcp.HealthCareProfessionals)
                    .Where(rel => rel.ClientId == clientId)
                    .ToList();

                to.HealthVisitor = GetHcpText(hcps.FirstOrDefault(hcp => hcp.StaffType == "HV"));
                to.Gp = GetHcpText(hcps.FirstOrDefault(hcp => hcp.StaffType == "GP"));

                to.IsWarning = uow.Repository<Warning>().AsNoTracking().Any(warning => warning.ClientId == clientId &&
                    warning.FkWarningType == (long)LookupsUtility.WarningType.WdsAddressOrGpUpdate &&
                    warning.Active == OVXBoolean.True);

                return to;
            }
        }
        public ClientSummaryModel GetClientSummary(long clientId)
        {
            using (var uow = unitOfWorkFactory())
            {
                DefaultAccessForChild(clientId);

                var child = uow.Repository<ChildGeneral>()
                    .Include(x => x.ChildCentres.Select(t => t.ChildLocations))
                    .Include(x => x.RelHcpDemographics.Select(t => t.HealthCareProfessionals))
                    .Include(x => x.ChildAlertCodes.Select(a => a.AlertCodes))
                    .Include(x => x.RemoteFlag)
                    .Where(x => x.ClientId == clientId)
                    .AsNoTracking()
                    .SingleOrDefault();

                var clientSummary = Mapper.Map<ChildGeneral, ClientSummaryModel>(child);

                if (child != null)
                {
                    clientSummary.Organisation = GetOrganisationValue(child.ChildCentres, uow);
                    clientSummary.RemoteChild = child.RemoteFlag.IsRemote == OVXBoolean.True;
                }
                
                return clientSummary;
            }
        }

        public void DefaultAccessForChild(long clientId)
        {
            using (var uow = unitOfWorkFactory())
            {
                AccessControlProvider.DefaultAccessForChild(clientId, uow);
            }
        }

        public void ResetDefaultAccess()
        {
            using (var uow = unitOfWorkFactory())
            {
                AccessControlProvider.ResetDefaultAccess(uow);
            }
        }

        private static string GetOrganisationValue(IEnumerable<ChildCentres> childCentres, IUnitOfWork uow)
        {
            string organisationValue;

            var locations = childCentres.FirstOrDefault(cc => cc.ChildLocationType == (byte)LookupsUtility.CENTRE_TYPES.HEALTH_AUTHORITY);
            if (locations != null)
            {
                if (locations.ChildLocations != null)
                {
                    var organisation = uow.Repository<ChildLocations>()
                        .AsNoTracking()
                        .SingleOrDefault(
                            cl => cl.Code == locations.ChildLocations.Organisation && cl.LocationTypes.Authority == "Y");

                    organisationValue = (organisation == null ? "-" : String.Join(", ", organisation.Code, organisation.Name));
                }
                else
                {
                    organisationValue = "-";
                }
            }
            else
            {
                organisationValue = "-";
            }

            return organisationValue;
        }

        private static string GetHcpText(RelHcpDemographics hcp)
        {
            if (hcp == null)
                return "-";

            return hcp.HealthCareProfessionals.Code + ",  " + hcp.HealthCareProfessionals.Name1 + " " + hcp.HealthCareProfessionals.Name2;
        }

        private static string GetPatientCentres(WisPatientCentre cc)
        {
            if (cc == null)
                return "-";

            return cc.FkLocationCode + ", " + cc.Location.Name;
        }

        private static string GetChildCentreText(ChildCentres cc)
        {
            if (cc == null)
                return "-";

            return cc.ChildLocationCode + ", " + cc.ChildLocations.Name;
        }
    }
}

using AutoMapper;
using Orvex.DomainModel;

namespace OrvexBusinessTierCommon.OVX.BusinessTier.Common.AutoMapperConfig.Resolvers
{
    public class AddressResolver : ValueResolver<AddressModel, string>
    {
        protected override string ResolveCore(AddressModel source)
        {
            var address = string.Empty; 

            if (!string.IsNullOrEmpty(source.Address1) && source.Address1 != "-")
                address += source.Address1 + ", ";
            if (!string.IsNullOrEmpty(source.Address2) && source.Address2 != "-")
                address += source.Address2 + ", ";
            if (!string.IsNullOrEmpty(source.Address3) && source.Address3 != "-")
                address += source.Address3 + ", ";
            if (!string.IsNullOrEmpty(source.Address4) && source.Address4 != "-")
                address += source.Address4 + ", ";
            if (!string.IsNullOrEmpty(source.Address5) && source.Address5 != "-")
                address += source.Address5 + ", ";
            if (!string.IsNullOrEmpty(source.Postcode) && source.Postcode != "-")
                address += source.Postcode;

            return address.TrimEnd().TrimEnd(',');
        }
    }
}

using System;
using AutoMapper;
using Orvex.Model;

namespace OrvexBusinessTierCommon.OVX.BusinessTier.Common.AutoMapperConfig.Resolvers
{
    public class AgeResolver : ValueResolver<ChildGeneral, string>
    {
        protected override string ResolveCore(ChildGeneral source)
        {
            return GetAge(source.DateOfBirth, source.DateOfDeath);
        }

        /// <summary>
        /// Ported from package LOOKUPS.AGE_DSP
        /// </summary>
        private static string GetAge(DateTime dateOfBirth, DateTime? thresholdDate)
        {
            string age;

            var dateToCalculate = DateTime.Now;

            if (thresholdDate.HasValue)
            {
                dateToCalculate = thresholdDate.Value;
            }

            var years = dateToCalculate.Year - dateOfBirth.Year;

            var months = (years*12) + (dateToCalculate.Month - dateOfBirth.Month);

            years = months / 12;

            if (years < 1)
            {
                if (months <= 3)
                {
                    var days = (int)(dateToCalculate - dateOfBirth).TotalDays;

                    if (days < 1)
                    {
                        age = "0 day";
                    }
                    else if (days == 1)
                    {
                        age = days + " day";
                    }
                    else if (days > 1 && days < 7)
                    {
                        age = days + " days";
                    }
                    else if (days < 14)
                    {
                        age = "1 week";
                    }
                    else
                    {
                        var weeks = (int)(dateToCalculate - dateOfBirth).TotalDays / 7;
                        age = weeks + " weeks";
                    }
                }
                else
                {
                    age = months + " months";
                }
            }
            else
            {
                months = months % 12;
                age = $"{years} Years {months} Months";
            }

            return age;
        }
    }
}

    {
        protected override string ResolveCore(INameModel source)
        {
            return (string.IsNullOrEmpty(source.Title) ? string.Empty : source.Title + " ")
                                   + (string.IsNullOrEmpty(source.Surname) ? string.Empty : source.Surname + ", ")

            ? ", " + CultureInfo.CurrentCulture.TextInfo.ToTitleCase(Person.Forename)
            : string.Empty);

        public string NhsNumber => Person.NhsNumber;

        public string NhsTraceResult
        {
            get
            {
                if (Person.NhsTraceResult != null)
                {
                    var lookup = lookupsDelegate.getLookupSpecific(TypeOfLookup.NhsTraceResult, Person.NhsTraceResult);
                    if (lookup.Any())
                    {
                        return CultureInfo.CurrentCulture.TextInfo.ToTitleCase(lookup.First().Description);
                    }
                }

                return string.Empty;
            }
        }

        public string Sex => LookupsUtility.Gender(false, LookupsUtility.DICTIONARY_TYPE.STRING)
            .First(g => g.Value == Person.Gender)
            .Key.ToUpper();

        public string Telephone => Person.Telephone;

        public string Mobile => Person.Mobile;

        private static string FormatAddress(string line1, string line2, params string[] otherAddressLines)
        {
            var str = new StringBuilder();

            if (!string.IsNullOrEmpty(line1))
            {
                str.Append(line1);
                str.Append(string.IsNullOrEmpty(line2)
                    ? Environment.NewLine
                    : ", ");
            }

            if (!string.IsNullOrEmpty(line2))
            {
                str.Append(line2);
                str.Append(Environment.NewLine);
            }

            foreach (var line in otherAddressLines.Where(l => !string.IsNullOrEmpty(l)))
            {
                str.Append(line);
                str.Append(Environment.NewLine);
            }

            return str.ToString();
        }

        {
            service.DefaultAccessForChild(clientId);
        }

        public void Close()
        {
            service.ResetDefaultAccess();
        }
    }

using System;
using System.Collections.Generic;
using System.Globalization;
using System.Linq;
using System.Linq.Expressions;
using System.Web;
using System.Web.Mvc;
using System.Web.Routing;

namespace Orvex.WebClient.Extensions
{
    public static class StaticSpan
    {
        public static MvcHtmlString SpanFor<TModel, TProperty>(this HtmlHelper<TModel> helper, Expression<Func<TModel, TProperty>> expression, object htmlAttributes = null, string format = null, bool leaveBlankIfNull = false)
        {
            var valueGetter = expression.Compile();
            var value = valueGetter(helper.ViewData.Model);


            var span = new TagBuilder("span");
            span.MergeAttributes(new RouteValueDictionary(htmlAttributes));

            if (value is DateTime? && value.ToString() == DateTime.MinValue.ToString(CultureInfo.InvariantCulture))
            {
                span.SetInnerText(leaveBlankIfNull ? string.Empty : "-");

                return MvcHtmlString.Create(span.ToString());
            }


            if (value != null && !string.IsNullOrEmpty(value.ToString()))
            {
                span.SetInnerText(format != null ? string.Format(format, value) : value.ToString());
            }
            else
            {
                span.SetInnerText(leaveBlankIfNull ? string.Empty : "-");
            }

            return MvcHtmlString.Create(span.ToString());
        }
    }
}

        {
            clientSearchDelegate.getTO("");

            PopulateDetails();
        }

        private void PopulateDetails()
        {
            personBindingSource.DataSource = personDelegate;
            
            switch (GlobalSessionValues.BusinessArea)
            {
                case GlobalSessionValues.BusinessAreas.ChildHealth:
                    uiGroupBoxDetails.Controls.Clear();
                    var ovxdets = new ucClientHeaderOVXDetails(personDelegate.Person);
                    uiGroupBoxDetails.Controls.Add(ovxdets);
                    ovxdets.Dock = DockStyle.Fill;
                    uiGroupBoxFurtherDetails.Controls.Clear();
                    var ovxfdets = new ucClientHeaderOVXFurtherDetails(personDelegate.Person);
                    ovxfdets.Dock = DockStyle.Fill;
                    uiGroupBoxFurtherDetails.Controls.Add(ovxfdets);

                    //Set Group Box Widths
                    uiGroupBoxDetails.Width = ovxdets.MaxGroupBoxLength;
                    uiGroupBoxFurtherDetails.Width = ovxfdets.MaxGroupBoxLength;
                    
                    break;

                case GlobalSessionValues.BusinessAreas.ChildProtection:
                    var cpdets = new ucClientHeaderCPDetails(personDelegate.Person);
                    cpdets.Dock = DockStyle.Fill;
                    uiGroupBoxDetails.Controls.Add(cpdets);
                    var cpfdets = new ucClientHeaderCPFurtherDetails(personDelegate.Person);
                    cpfdets.Dock = DockStyle.Fill;
                    uiGroupBoxFurtherDetails.Controls.Add(cpfdets);

                    //Set Group Box Widths
                    uiGroupBoxDetails.Width = cpdets.MaxGroupBoxLength;
                    break;

                case GlobalSessionValues.BusinessAreas.LookedAfterChildren:
                    var lacdets = new ucClientHeaderLACDetails(personDelegate.Person);
                    lacdets.Dock = DockStyle.Fill;
                    uiGroupBoxDetails.Controls.Add(lacdets);
                    var lacfdets = new ucClientHeaderLACFurtherDetails(personDelegate.Person);
                    lacfdets.Dock = DockStyle.Fill;
                    uiGroupBoxFurtherDetails.Controls.Add(lacfdets);

                    //Set Group Box Widths
                    uiGroupBoxDetails.Width = lacdets.MaxGroupBoxLength;
                    break;

                case GlobalSessionValues.BusinessAreas.EmotionalWellbeing:
                    var ewbdets = new ucClientHeaderEWBDetails(personDelegate.Person);
                    ewbdets.Dock = DockStyle.Fill;
                    uiGroupBoxDetails.Controls.Add(ewbdets);
                    var ewbfdets = new ucClientHeaderEWBFurtherDetails(personDelegate.Person);
                    ewbfdets.Dock = DockStyle.Fill;
                    uiGroupBoxFurtherDetails.Controls.Add(ewbfdets);

                    //Set Group Box Widths
                    uiGroupBoxDetails.Width = ewbdets.MaxGroupBoxLength;
                    break;

                default:
                    var ovxdets1 = new ucClientHeaderOVXDetails(personDelegate.Person);
                    ovxdets1.Dock = DockStyle.Fill;
                    uiGroupBoxDetails.Controls.Add(ovxdets1);
                    var ovxfdets1 = new ucClientHeaderOVXFurtherDetails(personDelegate.Person);
                    ovxfdets1.Dock = DockStyle.Fill;
                    uiGroupBoxFurtherDetails.Controls.Add(ovxfdets1);
                    break;
            }

            var newSearch = new ClientSearchSubTO {ClientId = personDelegate.Person.ClientId};
            clientSearchDelegate.AddClientSearch(newSearch);
            clientSearchDelegate.save();
            clientSearchDelegate.refresh();

            Application.DoEvents();
            Refresh();

            CalcMaxGroupBoxLength();

            if (!personDelegate.Person.Remote) return;
            SetRemotePatientTextColor();
        }

        private void SetRemotePatientTextColor()
        {
            if (personDelegate.Person.Remote)
            {
                labelFullName.ForeColor = Color.ForestGreen;
                labelNHSNo.ForeColor = Color.ForestGreen;
                labelAddress.ForeColor = Color.ForestGreen;
                labelSex.ForeColor = Color.ForestGreen;
                labelDOB.ForeColor = Color.ForestGreen;
                ovxLabelMobile.ForeColor = Color.ForestGreen;
                ovxLabelTelephone.ForeColor = Color.ForestGreen;
                labelTraceResult.ForeColor = Color.ForestGreen;
            }
        }

        private void uiGroupBoxAddress_Resize(object sender, EventArgs e)
        {
            if (uiGroupBoxAddress.Height > 0)
            {
                uiGroupBoxAddressDets.Top = Convert.ToInt32(((uiGroupBoxAddress.Height - uiGroupBoxAddressDets.Height)/2));
            }
        }

        private void CalcMaxGroupBoxLength()
        {
            int maxLabelWidth = 0;

            foreach (OvxLabel lbl in uiGroupBoxAddressDets.Controls)
            {
                lbl.AutoSize = true;
                if (lbl.Width > maxLabelWidth)
                    maxLabelWidth = lbl.Width;
                lbl.AutoSize = false;
            }

            ResizeAllLabels(maxLabelWidth);

            if (maxLabelWidth < 110)
            {
                uiGroupBoxAddress.Width = 130; //12 is padding
                uiGroupBoxAddressDets.Width = maxLabelWidth + 4;
            }
            else
            {
                uiGroupBoxAddress.Width = maxLabelWidth + 4 + 24; //12 is padding
                uiGroupBoxAddressDets.Width = maxLabelWidth + 4;
            }
        }

        private void ResizeAllLabels(int newLabelWidth)
        {
            foreach (OvxLabel lbl in uiGroupBoxAddressDets.Controls)
            {
                if (lbl.LabelType == OvxLabel.TypeOfLabel.DeveloperCustom2)
                {
                    lbl.Width = newLabelWidth;
                }
            }
        }

using System;
using Orvex.TransferObjects;

namespace OVX.Net2008
{
    // ReSharper disable once InconsistentNaming
    public partial class ucClientHeaderLACDetails : OVXUserControl
    {
        public int MaxGroupBoxLength = 0;

        public ucClientHeaderLACDetails(PersonTO lacEntity)
        {
            InitializeComponent();

            labelSerialNo.Text = lacEntity.ClientId.ToString("D");
            labelSurname.Text = lacEntity.Surname;
            labelForename.Text = lacEntity.Forename;
            labelAge.Text = lacEntity.AgeString;

            CalcMaxGroupBoxLength();
        }

        private void ovxGroupBoxDetails_Resize(object sender, EventArgs e)
        {
            if (ovxGroupBoxDetails.Height > 0)
            {
                uiGroupBoxDetails.Top = Convert.ToInt32(((ovxGroupBoxDetails.Height - uiGroupBoxDetails.Height) / 2));
            }
        }

        private void CalcMaxGroupBoxLength()
        {
            int maxLabelWidth = 0;

            foreach (OvxLabel lbl in uiGroupBoxDetails.Controls)
            {
                lbl.AutoSize = true;
                if (lbl.Width > maxLabelWidth)
                    maxLabelWidth = lbl.Width;
                lbl.AutoSize = false;
            }

            ResizeAllLabels(maxLabelWidth);

            //124 is label start location and 24 is padding allowance
            MaxGroupBoxLength = maxLabelWidth + 73 + 24;
            uiGroupBoxDetails.Width = MaxGroupBoxLength - 5; //24 is padding;                
        }

        private void ResizeAllLabels(int newLabelWidth)
        {
            foreach (OvxLabel lbl in uiGroupBoxDetails.Controls)
            {
                if (lbl.LabelType == OvxLabel.TypeOfLabel.DeveloperCustom2)
                {
                    lbl.Width = newLabelWidth;
                }
            }
        }

    }
}

using System;
using System.Drawing;
using OVX.BusinessTier.Common.Util;
using OVX.Net2008.System_Options_and_Navigation.Filter;
using Orvex.TransferObjects;

namespace OVX.Net2008
{
    // ReSharper disable once InconsistentNaming
    public partial class ucClientHeaderOVXDetails : OVXUserControl
    {
        public int MaxGroupBoxLength;

        public ucClientHeaderOVXDetails(PersonTO person)
        {
            InitializeComponent();

            PopulateOvxFurtherDetails(person);
        }

        /**
        * Because this page is an exception to the rule we add our own tab exclusion so we only process to the point of tabs
        */
        public override void setupAccessLevelFilter(ControlAccessLevelFiltererDecorator filter, System.Windows.Forms.Control control)
        {
            // exclude details tab from being processed according to the new menyu restrictions
            filter.getFilter().addActiveComponentExclusion(typeof(ucClientHeaderOVXDetails));
        }

        public void PopulateOvxFurtherDetails(PersonTO person)
        {
            if (person != null)
            {
                labelSerialNo.Text = person.ClientId.ToString("D");
                labelSurname.Text = person.Surname;
                labelForename.Text = person.Forename;
                var currentStatusDate = string.Empty;

                if (!string.IsNullOrEmpty(person.CurrentStatus))
                {
                    switch (person.CurrentStatus)
                    {
                        case "7":
                            currentStatusDate = person.DateOfBirth.ToShortDateString();
                            break;
                        case "5":
                            currentStatusDate = person.DateOfDeath?.ToShortDateString();
                            break;
                        default:
                            currentStatusDate = person.CurrentStatusDate.ToShortDateString();
                            break;
                    }
                }
                labelStatus.Text = $@"{LookupsUtility.CurrentStatusText(person.CurrentStatus)} ({currentStatusDate})";
                labelAge.Text = person.AgeString;
                CalcMaxGroupBoxLength();

                if (!person.Remote) return;
                SetRemotePatientTextColor();
            }
        }
        private void SetRemotePatientTextColor()
        {
            labelSerialNo.ForeColor = Color.ForestGreen;
            labelSurname.ForeColor = Color.ForestGreen;
            labelForename.ForeColor = Color.ForestGreen;
            labelStatus.ForeColor = Color.ForestGreen;
            labelAge.ForeColor = Color.ForestGreen;
        }

        private void uiGroupBoxDetails_Resize(object sender, EventArgs e)
        {
            if (uiGroupBoxDetails.Height > 0)
            {
                ovxGroupBoxDetails.Top = Convert.ToInt32(((uiGroupBoxDetails.Height - ovxGroupBoxDetails.Height) / 2));
            }
        }

        private void CalcMaxGroupBoxLength()
        {
            int maxLabelWidth = 0;

            foreach (OvxLabel lbl in ovxGroupBoxDetails.Controls)
            {
                lbl.AutoSize = true;
                if (lbl.Width > maxLabelWidth)
                    maxLabelWidth = lbl.Width;
                lbl.AutoSize = false;
            }

            ResizeAllLabels(maxLabelWidth);

            //124 is label start location and 24 is padding allowance
            MaxGroupBoxLength = maxLabelWidth + 73 + 24;
            ovxGroupBoxDetails.Width = MaxGroupBoxLength - 5; //24 is padding;                
        }

        private void ResizeAllLabels(int newLabelWidth)
        {
            foreach (OvxLabel lbl in ovxGroupBoxDetails.Controls)
            {
                if (lbl.LabelType == OvxLabel.TypeOfLabel.DeveloperCustom2)
                {
                    lbl.Width = newLabelWidth;
                }
            }
        }

    }

    // ReSharper disable once InconsistentNaming
    public partial class ucClientHeaderOVXFurtherDetails : OVXUserControl
    {
        public int MaxGroupBoxLength;
        private PersonTO person;
        public ucClientHeaderOVXFurtherDetails(PersonTO person)
        {
            InitializeComponent();

            PopulateOvxFurtherDetails(person);
        }

        private void PopulateOvxFurtherDetails(PersonTO person)
        {
            if (person != null)
            {
                this.person = person;
                labelGP.Text = person.Gp;
                labelHealthVisitor.Text = person.HealthVisitor;
                labelTreatmentCentre.Text = person.TreatmentCentre;
                labelExamCentre.Text = person.ExamCentre;
                labelSchool.Text = person.School;
                labelCareTrustCode.Text = person.Organisation;
                labelPracticeCode.Text = person.Practice;
                ovxSchoolImmLocDescLbl.Text = person.SchoolImmLoc;

                CalcMaxGroupBoxLength();

                if (!person.Remote) return;
                SetRemotePatientTextColor();           
            }
        }
       
        private void SetRemotePatientTextColor()
        {
            labelGP.ForeColor = Color.ForestGreen;
            labelHealthVisitor.ForeColor = Color.ForestGreen;
            labelTreatmentCentre.ForeColor = Color.ForestGreen;
            labelExamCentre.ForeColor = Color.ForestGreen;
            labelSchool.ForeColor = Color.ForestGreen;
            labelCareTrustCode.ForeColor = Color.ForestGreen;
            labelPracticeCode.ForeColor = Color.ForestGreen;
        }

        private void uiGroupBoxFurtherDetails_Resize(object sender, EventArgs e)
        {
            if (uiGroupBoxFurtherDetails.Height > 0)
            {
                ovxGroupBoxDetails.Top = Convert.ToInt32(((uiGroupBoxFurtherDetails.Height - ovxGroupBoxDetails.Height) / 2));
            }
        }

        private void CalcMaxGroupBoxLength()
        {
            int maxLabelWidth = 0;

            foreach (OvxLabel lbl in ovxGroupBoxDetails.Controls)
            {
                lbl.AutoSize = true;
                if (lbl.Width > maxLabelWidth)
                    maxLabelWidth = lbl.Width;
                lbl.AutoSize = false;
            }

            ResizeAllLabels(maxLabelWidth);

            //124 is label start location and 12 is padding allowance
            MaxGroupBoxLength = maxLabelWidth + 124 + 24;
            ovxGroupBoxDetails.Width = MaxGroupBoxLength - 2;
        }

        private void ResizeAllLabels(int newLabelWidth)
        {
            foreach (OvxLabel lbl in ovxGroupBoxDetails.Controls)
            {
                if (lbl.LabelType == OvxLabel.TypeOfLabel.DeveloperCustom2)
                {
                    lbl.Width = newLabelWidth;
                }
            }
        }

        #region Tool Tips
        private void labelHealthVisitor_MouseHover(object sender, EventArgs e)
        {
            toolTipFurtherDetails.SetToolTip(labelHealthVisitor, labelHealthVisitor.Text);
        }

        private void labelGP_MouseHover(object sender, EventArgs e)
        {

        }
